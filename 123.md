cmake_minimum_required(VERSION 3.16)
project(LibraryKR)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(PostgreSQL_ROOT /usr)
set(PostgreSQL_INCLUDE_DIR /usr/include)
set(PostgreSQL_LIBRARY /usr/lib/libpq.so)
find_package(PostreSQL REQUIRED)

add_executable(library_server
  server/main.c
  server/db.c
  server/handler.c
)

target_include_directories(library_server PRIVATE
  ${PostreSQL_INCLUDE_DIRS}
)

target_link_libraries(library_server PRIVATE
  ${PostreSQL_LIBRARIES}
)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(WEBVIEW_WEBKITGTK_API 6.0)	

include(FetchContent)

FetchContent_Declare(
    webview
    GIT_REPOSITORY https://github.com/webview/webview
    GIT_TAG 0.12.0
)
FetchContent_MakeAvailable(webview)

FetchContent_Declare(
  json
  GIT_REPOSITORY https://github.com/nlohmann/json 
  GIT_TAG v3.12.0
)
FetchContent_MakeAvailable(json)

add_executable(app WIN32)

target_sources(app PRIVATE src/main.cpp ui/dist/index_html.h)
target_include_directories(app PRIVATE ui/dist)
target_link_libraries(app PRIVATE webview::core nlohmann_json::nlohmann_json)
